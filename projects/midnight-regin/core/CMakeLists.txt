cmake_minimum_required(VERSION 3.16)
project(MidnightReignCore CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(mr_core INTERFACE)
target_include_directories(mr_core INTERFACE include)

enable_testing()
foreach(test_name core_tests race_tests)
  add_executable(mr_${test_name} tests/${test_name}.cpp)
  target_link_libraries(mr_${test_name} PRIVATE mr_core)
  if(MSVC)
    target_compile_options(mr_${test_name} PRIVATE /W4 /WX)
  else()
    target_compile_options(mr_${test_name} PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME mr_${test_name} COMMAND mr_${test_name})
endforeach()
